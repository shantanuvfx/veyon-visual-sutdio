<script>
  // ============ PARTICLE SYSTEM ============
  const canvas = document.getElementById('particleCanvas');
  const ctx = canvas.getContext('2d');

  function resizeCanvas() {
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
  }
  resizeCanvas();
  window.addEventListener('resize', resizeCanvas);

  const PARTICLE_COUNT = 120;
  const particles = [];

  const colors = [
    'rgba(123,92,250,',   // purple
    'rgba(77,184,255,',   // blue
    'rgba(162,89,255,',   // violet
    'rgba(200,160,255,',  // light purple
    'rgba(140,210,255,',  // light blue
    'rgba(255,255,255,',  // white sparkle
  ];

  class Particle {
    constructor() { this.reset(true); }
    reset(init) {
      this.x = Math.random() * canvas.width;
      this.y = init ? Math.random() * canvas.height : canvas.height + 10;
      this.size = Math.random() * 2.2 + 0.3;
      this.speedY = -(Math.random() * 0.4 + 0.1);
      this.speedX = (Math.random() - 0.5) * 0.25;
      this.opacity = Math.random() * 0.5 + 0.05;
      this.maxOpacity = this.opacity;
      this.color = colors[Math.floor(Math.random() * colors.length)];
      this.life = 0;
      this.maxLife = Math.random() * 400 + 200;
      // Some particles are tiny lines (streaks)
      this.isStreak = Math.random() < 0.2;
      this.streakLen = Math.random() * 12 + 4;
      this.angle = Math.random() * Math.PI * 2;
      // Twinkle
      this.twinkleSpeed = Math.random() * 0.02 + 0.005;
      this.twinkleOffset = Math.random() * Math.PI * 2;
      // Connection
      this.connected = false;
    }
    update() {
      this.x += this.speedX;
      this.y += this.speedY;
      this.life++;
      // Twinkle
      const twinkle = Math.sin(this.life * this.twinkleSpeed + this.twinkleOffset);
      this.currentOpacity = this.maxOpacity * (0.5 + 0.5 * twinkle);
      // Fade in/out
      const lifeRatio = this.life / this.maxLife;
      if (lifeRatio < 0.1) this.currentOpacity *= lifeRatio / 0.1;
      if (lifeRatio > 0.8) this.currentOpacity *= (1 - lifeRatio) / 0.2;
      if (this.life > this.maxLife || this.y < -20) this.reset(false);
    }
    draw() {
      ctx.save();
      if (this.isStreak) {
        ctx.globalAlpha = this.currentOpacity * 0.7;
        ctx.strokeStyle = this.color + '0.8)';
        ctx.lineWidth = this.size * 0.5;
        ctx.beginPath();
        ctx.moveTo(this.x, this.y);
        ctx.lineTo(
          this.x + Math.cos(this.angle) * this.streakLen,
          this.y + Math.sin(this.angle) * this.streakLen
        );
        ctx.stroke();
      } else {
        ctx.globalAlpha = this.currentOpacity;
        ctx.fillStyle = this.color + '1)';
        ctx.shadowColor = this.color + '0.8)';
        ctx.shadowBlur = this.size * 4;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        ctx.fill();
      }
      ctx.restore();
    }
  }

  for (let i = 0; i < PARTICLE_COUNT; i++) particles.push(new Particle());

  // Connection lines between nearby particles
  function drawConnections() {
    const maxDist = 100;
    for (let i = 0; i < particles.length; i++) {
      for (let j = i + 1; j < particles.length; j++) {
        const dx = particles[i].x - particles[j].x;
        const dy = particles[i].y - particles[j].y;
        const dist = Math.sqrt(dx*dx + dy*dy);
        if (dist < maxDist) {
          const opacity = (1 - dist / maxDist) * 0.07;
          ctx.save();
          ctx.globalAlpha = opacity;
          ctx.strokeStyle = 'rgba(123,92,250,1)';
          ctx.lineWidth = 0.4;
          ctx.beginPath();
          ctx.moveTo(particles[i].x, particles[i].y);
          ctx.lineTo(particles[j].x, particles[j].y);
          ctx.stroke();
          ctx.restore();
        }
      }
    }
  }

  function animParticles() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    drawConnections();
    particles.forEach(p => { p.update(); p.draw(); });
    requestAnimationFrame(animParticles);
  }
  animParticles();

  // Reveal
  const revObs = new IntersectionObserver(entries => entries.forEach(e => { if (e.isIntersecting) e.target.classList.add('visible'); }), { threshold: 0.1 });

  // Counter
  function animCount(el, target) {
    const suffix = el.dataset.target >= 10 ? '+' : 'M+';
    let start = null;
    (function step(ts) {
      if (!start) start = ts;
      const p = Math.min((ts - start) / 1600, 1);
      el.textContent = Math.floor(p * target) + suffix;
      if (p < 1) requestAnimationFrame(step);
    })(performance.now());
  }
  new IntersectionObserver(entries => {
    if (entries[0].isIntersecting) {
      document.querySelectorAll('.stat-num').forEach(n => animCount(n, +n.dataset.target));
    }
  }, { threshold: 0.3 }).observe(document.getElementById('stats'));
